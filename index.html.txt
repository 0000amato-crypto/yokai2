    let currentQuestions = [];
    let currentIndex = 0;
    let correctCount = 0;
    let sessionLog = [];
    let timerInterval = null;
    let timeLeft = 30;

    function getSCount() {
        return parseInt(localStorage.getItem('yokai2_s_count') || '0');
    }

    function addSCount() {
        let count = getSCount() + 1;
        localStorage.setItem('yokai2_s_count', count);
        return count;
    }

    function updateStartScreen() {
        document.getElementById('s-count-info').innerText = `現在のSランク達成回数: ${getSCount()}回`;
    }
    updateStartScreen();

    function startGame() {
        let shuffled = [...allQuestionsData].sort(() => 0.5 - Math.random());
        currentQuestions = shuffled.slice(0, 5);
        currentIndex = 0;
        correctCount = 0;
        sessionLog = [];

        switchScreen('quiz-screen');
        loadQuestion();
    }

    function switchScreen(screenId) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }

    function loadQuestion() {
        if (currentIndex >= currentQuestions.length) {
            showResult();
            return;
        }

        const q = currentQuestions[currentIndex];
        document.getElementById('question-progress').innerText = `第 ${currentIndex + 1} 問 / 5問中`;
        
        const nameBoxesContainer = document.getElementById('name-boxes');
        nameBoxesContainer.innerHTML = '';
        const cleanName = q.name.replace(/[^ぁ-んァ-ンー]/g, '');
        for (let i = 0; i < cleanName.length; i++) {
            const box = document.createElement('div');
            box.className = 'name-box';
            nameBoxesContainer.appendChild(box);
        }

        document.getElementById('clue-1').innerText = "〜 " + q.clues[0];
        document.getElementById('clue-2').innerText = "〜 " + q.clues[1];
        document.getElementById('clue-3').innerText = "〜 " + q.clues[2];

        const input = document.getElementById('answer-input');
        input.value = '';
        input.disabled = false;
        document.querySelector('.btn-submit').disabled = false;
        document.querySelector('.btn-giveup').disabled = false;
        document.getElementById('result-overlay').classList.remove('active');

        startTimer();
    }

    function startTimer() {
        clearInterval(timerInterval);
        timeLeft = 30;
        updateTimerDisplay();

        timerInterval = setInterval(() => {
            timeLeft--;
            updateTimerDisplay();

            if (timeLeft <= 0) {
                clearInterval(timerInterval);
                handleTimeOut();
            }
        }, 1000);
    }

    function updateTimerDisplay() {
        document.getElementById('timer-display').innerText = `残り ${timeLeft}秒`;
    }

    function submitAnswer() {
        const input = document.getElementById('answer-input');
        const userVal = input.value.trim().toLowerCase();
        if (!userVal || timeLeft <= 0) return;

        const q = currentQuestions[currentIndex];
        const isCorrect = q.accepted.map(a => a.toLowerCase()).includes(userVal);

        if (isCorrect) {
            clearInterval(timerInterval);
            correctCount++;
            sessionLog.push({ qNum: currentIndex + 1, name: q.name, status: '正解' });

            const boxes = document.querySelectorAll('.name-box');
            const cleanName = q.name.replace(/[^ぁ-んァ-ンー]/g, '');
            for (let i = 0; i < boxes.length && i < cleanName.length; i++) {
                boxes[i].innerText = cleanName[i];
            }

            showResultOverlay('correct', '正解！', `正解は、${q.name}！`);
            setTimeout(proceedToNext, 2000);
        } else {
            showResultOverlay('incorrect', '不正解！', 'もう一度入力してね！');
            setTimeout(() => {
                document.getElementById('result-overlay').classList.remove('active');
            }, 1000);
        }
    }

    document.getElementById('answer-input').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') submitAnswer();
    });

    function showAnswer() {
        clearInterval(timerInterval);
        const q = currentQuestions[currentIndex];
        sessionLog.push({ qNum: currentIndex + 1, name: q.name, status: '答えを見た(不正解扱い)' });

        const boxes = document.querySelectorAll('.name-box');
        const cleanName = q.name.replace(/[^ぁ-んァ-ンー]/g, '');
        for (let i = 0; i < boxes.length && i < cleanName.length; i++) {
            boxes[i].innerText = cleanName[i];
        }

        disableControls();
        showResultOverlay('timeout', '答えを見る', `正解は、${q.name}`);
        setTimeout(proceedToNext, 2500);
    }

    function handleTimeOut() {
        disableControls();
        const q = currentQuestions[currentIndex];
        sessionLog.push({ qNum: currentIndex + 1, name: q.name, status: '時間切れ(不正解扱い)' });

        const boxes = document.querySelectorAll('.name-box');
        const cleanName = q.name.replace(/[^ぁ-んァ-ンー]/g, '');
        for (let i = 0; i < boxes.length && i < cleanName.length; i++) {
            boxes[i].innerText = cleanName[i];
        }

        showResultOverlay('timeout', '時間切れ！', `正解は、${q.name}`);
        setTimeout(proceedToNext, 2500);
    }

    function disableControls() {
        document.getElementById('answer-input').disabled = true;
        document.querySelector('.btn-submit').disabled = true;
        document.querySelector('.btn-giveup').disabled = true;
    }

    function showResultOverlay(type, title, message) {
        const overlay = document.getElementById('result-overlay');
        const stamp = document.getElementById('result-stamp');
        const answerDisplay = document.getElementById('result-answer-display');

        stamp.className = `result-stamp ${type}`;
        stamp.innerText = title;
        answerDisplay.innerText = message;
        overlay.classList.add('active');
    }

    function proceedToNext() {
        currentIndex++;
        loadQuestion();
    }

    function showResult() {
        clearInterval(timerInterval);
        document.getElementById('final-stats-correct').innerText = `正解数: ${correctCount} / 5`;

        let rank = "Eランク妖怪";
        if (correctCount === 5) rank = "Sランク妖怪";
        else if (correctCount === 4) rank = "Aランク妖怪";
        else if (correctCount === 3) rank = "Bランク妖怪";
        else if (correctCount === 2) rank = "Cランク妖怪";
        else if (correctCount === 1) rank = "Dランク妖怪";
        else if (correctCount === 0) rank = "Eランク妖怪";

        let rankDisplayStr = `称号: ${rank}`;
        let legendMsg = "";

        if (correctCount === 5) {
            let sTotal = addSCount();
            if (sTotal >= 3) {
                rankDisplayStr = `称号: 🎉 レジェンド妖怪 🎉`;
                legendMsg = `【速報】Sランクを3回達成したため、最高峰の「レジェンド妖怪」の称号を手に入れました！おめでとうございます！`;
            } else {
                legendMsg = `Sランク達成！あと ${3 - sTotal} 回Sランクを取ると「レジェンド妖怪」に進化します！`;
            }
        }

        document.getElementById('final-rank-display').innerText = rankDisplayStr;
        document.getElementById('legend-message').innerText = legendMsg;

        const missedContainer = document.getElementById('missed-list-container');
        missedContainer.innerHTML = sessionLog.map(m => `第${m.qNum}問: ${m.name} [${m.status}]`).join('<br>');

        updateStartScreen();
        switchScreen('result-screen');
    }
